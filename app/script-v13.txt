var TN = null;
function tnLer(id) { return new RegExp('^[a-z0-9]{4,12}$').test(String(id || '')) ? lerProp('tn_' + id) : null; }
function tnUsar(c) { const t = tnLer(c); if (!t || t.on === false || !t.repo || !t.token) return false; TN = t; return true; }
function repoDe() { return TN ? TN.repo : REPO; }
function chaveDe() { return TN ? TN.token : GITHUB_TOKEN; }
function ch(k) { return TN ? k + '~' + TN.id : k; } // nome de propriedade ou de cache separado por treinador
function marcaDe() { return TN ? String(TN.nome || 'Consultoria') : 'AnabTech'; }
function hex(s) { return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, String(s), Utilities.Charset.UTF_8).map(function (b) { return ('0' + (b & 255).toString(16)).slice(-2); }).join(''); }
const COD_ABC = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789';
function codNovo() { const b = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, Utilities.getUuid() + ':' + Utilities.getUuid() + ':' + Date.now(), Utilities.Charset.UTF_8); let x = ''; for (let i = 0; i < 12; i++) x += COD_ABC.charAt((b[i] & 255) % 32); return x.slice(0, 4) + '-' + x.slice(4, 8) + '-' + x.slice(8); }
function codLimpo(c) { return String(c || '').toUpperCase().split('').filter(function (x) { return COD_ABC.indexOf(x) >= 0; }).join(''); }
function lerGit(caminho, repo, chave) {
  const r = UrlFetchApp.fetch('https://api.github.com/repos/' + repo + '/contents/' + caminho, { muteHttpExceptions: true, headers: { Authorization: 'Bearer ' + chave, Accept: 'application/vnd.github.raw', 'X-GitHub-Api-Version': '2022-11-28' } });
  if (r.getResponseCode() !== 200) return null;
  try { return JSON.parse(r.getContentText('UTF-8')); } catch (err) { return null; }
}
function listarGit(pasta, repo, chave) {
  const r = UrlFetchApp.fetch('https://api.github.com/repos/' + repo + '/contents/' + pasta, { muteHttpExceptions: true, headers: { Authorization: 'Bearer ' + chave, Accept: 'application/vnd.github+json', 'X-GitHub-Api-Version': '2022-11-28' } });
  if (r.getResponseCode() !== 200) return [];
  try { const L = JSON.parse(r.getContentText('UTF-8')); return Array.isArray(L) ? L.filter(function (x) { return x && x.type === 'file' && new RegExp('[.]json$').test(x.name); }).map(function (x) { return x.path; }) : []; } catch (err) { return []; }
}
// o treinador entra no app com o código de acesso que o dono gerou: recebe o endereço e a chave do repositório dele
function entrar(d) {
  const c = CacheService.getScriptCache(), f = Number(c.get('ent_f') || 0);
  if (f >= 12) return { erro: 'bloqueado' };
  const cod = codLimpo(d.cod), falha = function () { c.put('ent_f', String(f + 1), 900); return { erro: 'codigo' }; };
  if (cod.length !== 12) return falha();
  const h = hex('cod:' + cod), P = PropertiesService.getScriptProperties().getProperties();
  let t = null;
  Object.keys(P).forEach(function (k) { if (k.indexOf('tn_') !== 0) return; try { const x = JSON.parse(P[k]); if (x && x.h === h) t = x; } catch (err) {} });
  if (!t || t.on === false) return falha();
  t.ult = new Date().toISOString(); gravarProp('tn_' + t.id, t);
  return { ok: true, c: t.id, repo: t.repo, token: t.token, nome: t.nome, coach: t.coach || '' };
}
// cópia inicial para o treinador novo: os modelos de treino, a biblioteca e os padrões de dieta do dono. Nenhum aluno vai junto.
function semente(d) {
  if (!TN || !treinadorOk(d)) return { erro: 'auth' };
  const ler = function (pasta) { return listarGit(pasta, REPO, GITHUB_TOKEN).slice(0, 40).map(function (p) { return lerGit(p, REPO, GITHUB_TOKEN); }).filter(function (x) { return x && x.id; }); };
  const cfg = lerGit('config.json', REPO, GITHUB_TOKEN) || {};
  return { ok: true, modelos: ler('modelos'), bib: ler('biblioteca'), sugDieta: cfg.sugDieta || null, sugBase: cfg.sugBase || null };
}

// só o dono: criar, listar e mudar os treinadores
function tnCriar(d) {
  if (TN || !treinadorOk(d)) return { erro: 'auth' };
  const nome = String(d.nome || '').trim().slice(0, 40), coach = String(d.coach || '').trim().slice(0, 60), repo = String(d.repo || '').trim(), chave = String(d.chave || '').trim();
  if (!nome) return { erro: 'nome' };
  if (!new RegExp('^[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+$').test(repo) || repo.toLowerCase() === REPO.toLowerCase()) return { erro: 'repo' };
  if (chave.length < 20 || chave === GITHUB_TOKEN) return { erro: 'chave' };
  const r = UrlFetchApp.fetch('https://api.github.com/repos/' + repo, { muteHttpExceptions: true, headers: { Authorization: 'Bearer ' + chave, Accept: 'application/vnd.github+json', 'X-GitHub-Api-Version': '2022-11-28' } });
  if (r.getResponseCode() !== 200) return { erro: 'acesso', cod: r.getResponseCode() };
  let info = {}; try { info = JSON.parse(r.getContentText()); } catch (err) {}
  if (!info.private) return { erro: 'publico' };
  if (info.permissions && !info.permissions.push) return { erro: 'escrita' };
  const P = PropertiesService.getScriptProperties().getProperties();
  if (Object.keys(P).some(function (k) { if (k.indexOf('tn_') !== 0) return false; try { return String(JSON.parse(P[k]).repo).toLowerCase() === repo.toLowerCase(); } catch (err) { return false; } })) return { erro: 'repetido' };
  let id = ''; do { id = 't' + hex(Utilities.getUuid()).slice(0, 5); } while (P['tn_' + id]);
  const cod = codNovo();
  gravarProp('tn_' + id, { id: id, nome: nome, coach: coach, repo: repo, token: chave, h: hex('cod:' + codLimpo(cod)), ia: d.ia !== false, on: true, em: new Date().toISOString() });
  return { ok: true, id: id, cod: cod };
}
function tnListar(d) {
  if (TN || !treinadorOk(d)) return { erro: 'auth' };
  const P = PropertiesService.getScriptProperties().getProperties(), L = [];
  Object.keys(P).forEach(function (k) { if (k.indexOf('tn_') !== 0) return; try { const t = JSON.parse(P[k]); L.push({ id: t.id, nome: t.nome, coach: t.coach || '', repo: t.repo, ia: t.ia !== false, on: t.on !== false, em: t.em || '', ult: t.ult || '' }); } catch (err) {} });
  L.sort(function (a, b) { return String(a.em).localeCompare(String(b.em)); });
  return { ok: true, lista: L };
}
function tnMudar(d) {
  if (TN || !treinadorOk(d)) return { erro: 'auth' };
  const t = tnLer(d.id); if (!t) return { erro: 'id' };
  if (d.apagar === true) { gravarProp('tn_' + t.id, null); return { ok: true, apagado: true }; }
  if (typeof d.on === 'boolean') t.on = d.on;
  if (typeof d.ia === 'boolean') t.ia = d.ia;
  if (d.nome) t.nome = String(d.nome).trim().slice(0, 40);
  let cod = '';
  if (d.novoCod === true) { cod = codNovo(); t.h = hex('cod:' + codLimpo(cod)); }
  gravarProp('tn_' + t.id, t);
  return cod ? { ok: true, cod: cod } : { ok: true };
}

function doPost(e) {
  try {
    const raw = (e && e.postData && e.postData.contents) || '';
    if (!raw || raw.length > 15000000) return resposta('invalido');
    // só o arquivo do plano anterior (PDF ou foto) passa de 2 MB
    if (raw.length > 2200000 && raw.slice(0, 40).indexOf('"acao":"anexo"') < 0) return resposta('invalido');
    const d = JSON.parse(raw);
    if (d && d.c && !tnUsar(d.c)) return (!d.acao || d.acao === 'foto' || d.acao === 'registro') ? resposta('link') : json({ erro: 'link' });
    if (d && d.acao === 'entrar') return json(entrar(d));
    if (d && d.acao === 'semente') return json(semente(d));
    if (d && d.acao === 'tnCriar') return json(tnCriar(d));
    if (d && d.acao === 'tnListar') return json(tnListar(d));
    if (d && d.acao === 'tnMudar') return json(tnMudar(d));
    if (d && d.acao === 'foto') return resposta(receberFoto(d));
    if (d && d.acao === 'ia') return json(estimarRefeicao(d));
    if (d && d.acao === 'resumo') return json(resumirSemana(d));
    if (d && d.acao === 'anexo') return json(receberAnexo(d));
    if (d && d.acao === 'importar') return json(importarPlano(d));
    if (d && d.acao === 'push') return json(pushAlunoSalvar(d));
    if (d && d.acao === 'pushTreinador') return json(pushTreinador(d));
    if (d && d.acao === 'pushAluno') return json(pushParaAluno(d));
    if (d && d.acao === 'descanso') return json(descansoAvisar(d));
    if (raw.length > 150000) return resposta('invalido');
    if (d && d.acao === 'registro') return resposta(registrar(d));
    const pk = limpar(d);
    if (!pk) return resposta('invalido');
    const codigo = gravar(pk);
    if (codigo === 201) avisar('', 'Anamnese nova', String(pk.respostas.q_nome).slice(0, 60) + ' respondeu a anamnese.', 'Novo cadastro', String(pk.respostas.q_nome).slice(0, 60), '', '');
    return resposta(String(codigo));
  } catch (err) { return resposta('erro'); }
}
function doGet(e) {
  const q = (e && e.parameter) || {};
  // na URL o treinador vem em "tr": o Google reserva os parâmetros "c" e "sid" nos apps da web
  if (q.tr && !tnUsar(q.tr)) return json({ erro: 'link' });
  if (q.acao === 'versao') return json({ versao: VERSAO, ia: !!iaChave(), push: !!fcmConta(), aviso: JSON.parse(CacheService.getScriptCache().get(ch('ultAviso')) || 'null') });
  if (q.acao === 'marca') return json(marcaPublica());
  if (q.acao === 'testeaviso') return json(testeAviso());
  if (q.acao === 'aluno') return json(abrirPagina(q.t, q.p));
  return resposta('AnabTech: funcionando');
}

// ---------- anamnese ----------
function limpar(d) {
  if (!d || d.tipo !== 'mgcoach_anamnese_v2' || !/^[a-z0-9]{6,40}$/.test(String(d.id || ''))) return null;
  const txt = function (v, n) { return String(v == null ? '' : v).slice(0, n || 3000); };
  const respostas = {};
  Object.keys(d.respostas || {}).slice(0, 120).forEach(function (k) {
    if (!/^q_[A-Za-z0-9]+$/.test(k)) return;
    const v = d.respostas[k];
    respostas[k] = Array.isArray(v) ? v.slice(0, 40).map(function (x) { return txt(x, 300); }) : txt(v);
  });
  if (!String(respostas.q_nome || '').trim()) return null;
  const secoes = (Array.isArray(d.secoes) ? d.secoes : []).slice(0, 20).map(function (s) {
    return { t: txt(s && s.t, 120), itens: (Array.isArray(s && s.itens) ? s.itens : []).slice(0, 80).map(function (i) { return [txt(i && i[0], 300), txt(i && i[1])]; }) };
  });
  // plano anterior: arquivos que a própria anamnese guardou em anexos/an<id>/ e texto colado
  const anexos = (Array.isArray(d.anexos) ? d.anexos : []).slice(0, 8).filter(function (f) { return f && caminhoAnexo(f.p) && String(f.p).indexOf('anexos/an' + d.id + '/') === 0; })
    .map(function (f) { return { p: String(f.p), nome: txt(f.nome, 80), tipo: f.tipo === 'pdf' ? 'pdf' : 'jpg', kb: Math.min(99999, Math.round(Number(f.kb) || 0)) }; });
  const pk = { tipo: 'mgcoach_anamnese_v2', id: d.id, enviadaEm: txt(d.enviadaEm, 40), recebidaEm: new Date().toISOString(), respostas: respostas, secoes: secoes };
  if (anexos.length) pk.anexos = anexos;
  if (String(d.planoTxt || '').trim()) pk.planoTxt = txt(d.planoTxt, 30000);
  return pk;
}

function iaChave() { if (TN && TN.ia === false) return ''; return PropertiesService.getScriptProperties().getProperty('ANTHROPIC_KEY') || ''; }
function iaContar(t, max) {
  const P = PropertiesService.getScriptProperties(), dia = Utilities.formatDate(new Date(), 'America/Bahia', 'yyyyMMdd');
  const lk = LockService.getScriptLock(); lk.waitLock(5000);
  try {
    let C = {}; try { C = JSON.parse(P.getProperty('iaUso') || '{}'); } catch (err) {}
    if (C.d !== dia) C = { d: dia, tot: 0, al: {} };
    const k = t === 'treinador' ? ch(t) : idAluno(t);
    if ((C.al[k] || 0) >= (max || 20)) return 'aluno';
    if (C.tot >= 400) return 'total';
    C.al[k] = (C.al[k] || 0) + 1; C.tot++;
    P.setProperty('iaUso', JSON.stringify(C));
    return '';
  } finally { lk.releaseLock(); }
}

    }

      let validos = s.tk;
      msgs.slice(0, 2).forEach(function (m) { m.u = linkAluno(s.t, s.c || ''); const r = pushEnviar(validos, m); validos = r.validos; });
      if (validos.length !== s.tk.length) { s.tk = validos; P.setProperty(k, JSON.stringify(s)); }
    }
    // grava só o que mudou e relê antes, para não apagar uma refeição marcada enquanto este laço rodava
    if (JSON.stringify(st) !== (tudo['st_' + id] || '')) {
      const atual = lerProp('st_' + id);

    }
  });
}

// o app chama no início do descanso; a execução espera até o fim e manda a notificação, a menos que o descanso tenha sido pulado ou mudado
function descansoAvisar(d) {
  const t = String(d.t || ''), r = conferir(t, d.p); if (r.erro) return { erro: r.erro };
  const id = idAluno(t), C = CacheService.getScriptCache(), k = 'dsc_' + id, rid = String(d.rid || '').slice(0, 40);
  if (d.cancelar) { C.put(k, 'x', 600); return { ok: true, cancelado: true }; }
  const fim = Number(d.fim), espera = fim - Date.now();
  if (!/^[a-z0-9]{4,40}$/.test(rid) || !(espera > 2000) || espera > 300500) return { erro: 'tempo' };
  const s = lerProp('ps_' + id); if (!s || !s.tk || !s.tk.length) return { ok: true, sem: true };
  C.put(k, rid, 900);
  Utilities.sleep(Math.min(299000, Math.max(0, espera - 1200)));

  const res = pushEnviar(s.tk, { t: 'Acabou o descanso', b: prox ? 'Agora: ' + prox : 'Volte para a próxima série.', u: linkAluno(t), g: 'anab-descanso' });

function lerPagina(t) {
  if (!/^[a-f0-9]{32}$/.test(String(t || ''))) return null;
  return lerGit('portal/' + t + '.json', repoDe(), chaveDe());
}
function codigoHash(t, p) {
  return Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, t + ':' + p, Utilities.Charset.UTF_8)
    .map(function (b) { return ('0' + (b & 255).toString(16)).slice(-2); }).join('');
}
function conferir(t, p, pagina) {
  const c = CacheService.getScriptCache(), kf = 'f_' + t, falhas = Number(c.get(kf) || 0);
  if (falhas >= 8) return { erro: 'bloqueado' };
  let pin = c.get('h_' + t), ativo = c.get('a_' + t);
  if (!pin || pagina) {
    pagina = pagina || lerPagina(t);
    if (!pagina) return { erro: 'link' };
    pin = pagina.pin; ativo = pagina.ativo ? '1' : '0';
    c.put('h_' + t, pin, 600); c.put('a_' + t, ativo, 600); c.put('nm_' + t, String(pagina.nome || '').slice(0, 40), 900);
  }
  if (ativo !== '1') return { erro: 'link' };
  if (!/^[0-9]{4}$/.test(String(p || '')) || codigoHash(t, String(p)) !== pin) { c.put(kf, String(falhas + 1), 900); return { erro: 'codigo' }; }
  return { ok: true, pagina: pagina };
}
function abrirPagina(t, p) {
  t = String(t || '');
  const pagina = lerPagina(t);
  if (!pagina) return { erro: 'link' };
  const r = conferir(t, p, pagina);
  if (r.erro) return r;
  delete pagina.pin;
  return { ok: true, dados: pagina };
}
function registrar(d) {
  const t = String(d.t || '');
  const r = conferir(t, d.p);
  if (r.erro) return r.erro;
  const reg = d.reg;
  if (!reg || ['treino', 'dieta', 'feedback', 'pagamento'].indexOf(reg.tipo) < 0 || JSON.stringify(reg).length > 40000) return 'invalido';
  const c = CacheService.getScriptCache(), kn = 'n_' + t, n = Number(c.get(kn) || 0);
  if (n >= 150) return 'limite';
  c.put(kn, String(n + 1), 3600);
  const arq = t + '.' + Date.now().toString(36) + Math.random().toString(36).slice(2, 7);
  const codigo = salvar('portal-in/' + arq + '.json', { tipo: 'portal_reg', t: t, recebidoEm: new Date().toISOString(), reg: reg }, 'Registro do aluno');
  if (codigo !== 201) return 'erro';
  const nome = c.get('nm_' + t) || 'Um aluno', alvo = idAluno(t);
  const temTxt = function (v) { return String(v == null ? '' : v).split(' ').join('').length > 1; };
  if (reg.tipo === 'treino') {

  }

    avisar(t + 'fb', nome + (com ? ' mandou o feedback com um comentário' : ' mandou o feedback'), 'Abra o app para ler o feedback da semana.', com ? 'Feedback com comentário' : 'Feedback recebido', nome, 'feedback', alvo);

  else if (reg.tipo === 'dieta') {
    const dia = String(reg.data || ''), nf = Array.isArray(reg.feitas) ? reg.feitas.length : 0, tot = Number(reg.total) || 0;
    if (reg.nota) {
      // observação na dieta: avisa uma vez por texto (a dieta é reenviada a cada refeição marcada)
      const kd = 'dn_' + t + dia + String(reg.nota).length;
      if (!c.get(kd)) { c.put(kd, '1', 21600); avisar(kd, nome + ' deixou uma observação na dieta', 'Abra o app para ler e responder.', 'Observação na dieta', nome, 'obs', alvo); }

      const k1 = 'dd1_' + t + dia, k2 = 'ddc_' + t + dia;
      if (tot > 0 && nf >= tot) { if (!c.get(k2)) avisar(k2, nome + ' completou a dieta do dia', 'Todas as refeições foram marcadas. Abra o app para ver.', 'Dieta completa', nome, 'dieta', alvo); c.put(k2, '1', 21600); c.put(k1, '1', 21600); }

}

// O canal do ntfy, o e-mail extra, a cópia no Gmail e os tipos de aviso ligados ficam em config.json
// (ntfy, avisoEmail, avisoGmail, avisoTipos). Quem muda é o app, na aba Backup.
function lerAvisos(c) {
  let v = c.get(ch('avcfg'));
  if (v === null) {
    v = '{}';
    const j = lerGit('config.json', repoDe(), chaveDe());
    if (j) { try { v = JSON.stringify({ ntfy: String(j.ntfy || ''), email: String(j.avisoEmail || ''), gmail: j.avisoGmail !== false, tipos: (j.avisoTipos && typeof j.avisoTipos === 'object') ? j.avisoTipos : {}, mudos: Array.isArray(j.avisoMudos) ? j.avisoMudos.map(String) : [] }); } catch (err) {} }
    c.put(ch('avcfg'), v, 300);
  }
  return JSON.parse(v);
}
// Ordem pensada para a velocidade: 1) e-mail extra (assistente do WhatsApp), 2) cópia no Gmail, 3) ntfy por último,
// porque o ntfy às vezes demora a responder para os servidores do Google e não pode segurar os e-mails.
// No e-mail extra: assunto = tipo do evento, corpo = nome do aluno (formato pedido pelo assistente).
// cat = tipo de aviso que o treinador pode desligar no app; alvo = identificador curto para abrir o app direto no aluno.
function avisar(chave, titulo, texto, tipo, aluno, cat, alvo) {
  const res = { v: VERSAO, em: new Date().toISOString(), ntfy: 0, ms: {} };
  const link = '\n\nhttps://anabtech.github.io/app/';
  const quando = Utilities.formatDate(new Date(), 'America/Bahia', 'dd/MM/yyyy HH:mm:ss');
  const marca = function (k, t0) { res.ms[k] = Date.now() - t0; };
  try {
    const c = CacheService.getScriptCache(), cfg = lerAvisos(c);
    const canal = /^[A-Za-z0-9_-]{12,64}$/.test(cfg.ntfy) ? cfg.ntfy : '';
    // os e-mails saem da conta do dono do script: para os outros treinadores o aviso vai só por notificação e ntfy
    const extra = !TN && /^[^ @,;]+@[^ @,;]+[.][^ @,;]+$/.test(cfg.email) ? cfg.email : '';
    const gmail = !TN && cfg.gmail !== false;
    const temPush = (lerProp(ch('pt')) || []).length > 0;
    if (!canal && !extra && !gmail && !temPush) return { desligado: true };
    if (cat && cfg.tipos && cfg.tipos[cat] === false) return { desligado: true };
    if (alvo && Array.isArray(cfg.mudos) && cfg.mudos.indexOf(alvo) >= 0) return { desligado: true }; // perfil próprio do treinador: sem aviso
    const abre = 'https://anabtech.github.io/app/' + (alvo ? '#a=' + alvo : '');
    // perto do limite diário de e-mails do Google: guarda o que resta para o e-mail do assistente
    let pouca = false; try { pouca = MailApp.getRemainingDailyQuota() < 8; } catch (err) {}
    if (chave) { if (c.get('av_' + chave)) return { repetido: true }; c.put('av_' + chave, '1', 1200); }
    if (temPush) { try { res.push = pushParaTreinador(titulo, texto, abre); } catch (err) { res.pushErro = String(err).slice(0, 120); } }
    let t0 = Date.now();
    if (extra) {
      try { MailApp.sendEmail({ to: extra, name: 'AnabTech', subject: tipo || titulo, body: 'Aluno: ' + (aluno || '-') + '\nEvento: ' + (tipo || titulo) + '\nQuando: ' + quando + '\n\n' + titulo + '. ' + texto + link }); res.extra = extra.replace(/^(.{2})[^@]*@/, '$1…@'); }
      catch (err) { res.extraErro = String(err).slice(0, 160); }
      marca('extra', t0);
    }
    if ((gmail && !(pouca && (canal || (extra && !res.extraErro)))) || (!extra && !canal) || res.extraErro) {
      t0 = Date.now();
      try {
        const eu = Session.getEffectiveUser().getEmail();
        MailApp.sendEmail({ to: eu, name: 'AnabTech', subject: 'AnabTech · ' + titulo, body: texto + '\nQuando: ' + quando + '\n\n' + abre });
        res.email = eu.replace(/^(.{2})[^@]*@/, '$1…@');
      } catch (err) { res.emailErro = String(err).slice(0, 160); }
      marca('gmail', t0);
    }
    if (canal) {
      t0 = Date.now();
      try {
        const r = UrlFetchApp.fetch('https://ntfy.sh/', { method: 'post', contentType: 'application/json', muteHttpExceptions: true,
          payload: JSON.stringify({ topic: canal, title: titulo, message: texto, tags: ['muscle'], click: abre }) });
        res.ntfy = r.getResponseCode();
        if (res.ntfy !== 200) res.erro = String(r.getContentText()).slice(0, 160);
      } catch (err) { res.ntfy = -1; res.erro = String(err).slice(0, 160); }
      marca('ntfy', t0);
      // ntfy falhou e nenhum e-mail saiu: manda para o Gmail de reserva
      if (res.ntfy !== 200 && !res.extra && !res.email && !res.emailErro) {
        try { const eu = Session.getEffectiveUser().getEmail(); MailApp.sendEmail({ to: eu, name: 'AnabTech', subject: 'AnabTech · ' + titulo, body: texto + '\nQuando: ' + quando + link }); res.email = eu.replace(/^(.{2})[^@]*@/, '$1…@'); }
        catch (err) { res.emailErro = String(err).slice(0, 160); }
      }
    }
    try { res.cota = MailApp.getRemainingDailyQuota(); } catch (err) {}
    c.put(ch('ultAviso'), JSON.stringify(res), 21600);
  } catch (err) { res.erro = String(err).slice(0, 160); }
  return res;
}

// ---------- GitHub ----------
function salvar(caminho, obj, mensagem) { return enviar(caminho, Utilities.base64Encode(JSON.stringify(obj), Utilities.Charset.UTF_8), mensagem); }
function enviar(caminho, base64, mensagem) {
  const r = UrlFetchApp.fetch('https://api.github.com/repos/' + repoDe() + '/contents/' + caminho, {
    method: 'put', contentType: 'application/json', muteHttpExceptions: true,
    headers: { Authorization: 'Bearer ' + chaveDe(), Accept: 'application/vnd.github+json', 'X-GitHub-Api-Version': '2022-11-28' },
    payload: JSON.stringify({ message: mensagem, content: base64 })
  });
  return r.getResponseCode();
}
